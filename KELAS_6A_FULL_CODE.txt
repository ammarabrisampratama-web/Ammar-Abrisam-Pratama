==================== FILE: firestore.rules ====================
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Global safety net
    match /{document=**} {
      allow read, write: if false;
    }

    // --- Reusable Hardened Helpers ---
    function isValidId(id) {
      return id is string && id.size() >= 1 && id.size() <= 128 && id.matches('^[a-zA-Z0-9_\\-]+$');
    }

    function incoming() {
      return request.resource.data;
    }

    function existing() {
      return resource.data;
    }

    function isSignedIn() {
      return request.auth != null && request.auth.token.email_verified == true;
    }

    function isSelf(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    function isBootstrappedOwner() {
      return isSignedIn() && request.auth.token.email == 'ammarabrisampratama@gmail.com';
    }

    function hasPublicProfile() {
      return exists(/databases/$(database)/documents/users_public/$(request.auth.uid));
    }

    function getUserRole() {
      return get(/databases/$(database)/documents/users_public/$(request.auth.uid)).data.role;
    }

    function isOwner() {
      return isBootstrappedOwner() || (isSignedIn() && hasPublicProfile() && getUserRole() == 'owner');
    }

    function isAdminOrOwner() {
      return isBootstrappedOwner() || (isSignedIn() && hasPublicProfile() && getUserRole() in ['owner', 'admin']);
    }

    function isValidDateStr(d) {
      return d is string && d.size() == 10 && d.matches('^\\d{4}-\\d{2}-\\d{2}$');
    }

    function isValidTimeStr(t) {
      return t is string && t.size() == 5 && t.matches('^\\d{2}:\\d{2}$');
    }

    // --- Validation Blueprints ---
    function isValidUserPublic(data) {
      return data.keys().hasAll(['uid', 'displayName', 'absenNumber', 'role', 'classId', 'updatedAt'])
        && data.keys().hasOnly(['uid', 'displayName', 'absenNumber', 'role', 'classId', 'updatedAt'])
        && isValidId(data.uid)
        && data.displayName is string && data.displayName.size() >= 2 && data.displayName.size() <= 80
        && data.absenNumber is int && data.absenNumber >= 0 && data.absenNumber <= 60
        && data.role in ['owner', 'admin', 'siswa']
        && data.classId == '6A'
        && data.updatedAt == request.time;
    }

    function isValidUserPrivate(data) {
      return data.keys().hasAll(['uid', 'email', 'createdAt', 'lastLoginAt'])
        && data.keys().hasOnly(['uid', 'email', 'createdAt', 'lastLoginAt'])
        && isValidId(data.uid)
        && data.email is string && data.email.size() >= 3 && data.email.size() <= 160
        && data.createdAt is timestamp
        && data.lastLoginAt == request.time;
    }

    function isValidTaskProgress(data) {
      return data.keys().hasAll(['taskId', 'uid', 'completed', 'updatedAt'])
        && data.keys().hasOnly(['taskId', 'uid', 'completed', 'updatedAt'])
        && isValidId(data.taskId)
        && isValidId(data.uid)
        && data.completed is bool
        && data.updatedAt == request.time;
    }

    function isValidLoginHistory(data) {
      return data.keys().hasAll(['uid', 'displayName', 'email', 'role', 'deviceInfo', 'loggedInAt'])
        && data.keys().hasOnly(['uid', 'displayName', 'email', 'role', 'deviceInfo', 'loggedInAt'])
        && isValidId(data.uid)
        && data.uid == request.auth.uid
        && data.displayName is string && data.displayName.size() >= 1 && data.displayName.size() <= 80
        && data.email is string && data.email.size() >= 3 && data.email.size() <= 160
        && data.role in ['owner', 'admin', 'siswa']
        && data.deviceInfo is string && data.deviceInfo.size() >= 1 && data.deviceInfo.size() <= 160
        && data.loggedInAt == request.time;
    }

    function isValidAnnouncement(data) {
      return data.keys().hasAll(['classId', 'title', 'content', 'category', 'authorId', 'authorName', 'authorRole', 'createdAt'])
        && data.keys().hasOnly(['classId', 'title', 'content', 'category', 'authorId', 'authorName', 'authorRole', 'createdAt'])
        && data.classId == '6A'
        && data.title is string && data.title.size() >= 3 && data.title.size() <= 120
        && data.content is string && data.content.size() >= 3 && data.content.size() <= 2000
        && data.category in ['Penting', 'Akademik', 'Kegiatan', 'Umum']
        && isValidId(data.authorId)
        && data.authorName is string && data.authorName.size() >= 2 && data.authorName.size() <= 80
        && data.authorRole in ['owner', 'admin']
        && data.createdAt is timestamp;
    }

    function isValidAssignment(data) {
      return data.keys().hasAll(['classId', 'subject', 'title', 'description', 'dueDate', 'authorId', 'authorName', 'createdAt', 'updatedAt'])
        && data.keys().hasOnly(['classId', 'subject', 'title', 'description', 'dueDate', 'authorId', 'authorName', 'createdAt', 'updatedAt'])
        && data.classId == '6A'
        && data.subject is string && data.subject.size() >= 2 && data.subject.size() <= 60
        && data.title is string && data.title.size() >= 3 && data.title.size() <= 120
        && data.description is string && data.description.size() >= 1 && data.description.size() <= 1500
        && isValidDateStr(data.dueDate)
        && isValidId(data.authorId)
        && data.authorName is string && data.authorName.size() >= 2 && data.authorName.size() <= 80
        && data.createdAt is timestamp
        && data.updatedAt == request.time;
    }

    function isValidScheduleItem(data) {
      return data.keys().hasAll(['classId', 'day', 'startTime', 'endTime', 'subject', 'teacher', 'room', 'authorId', 'updatedAt'])
        && data.keys().hasOnly(['classId', 'day', 'startTime', 'endTime', 'subject', 'teacher', 'room', 'authorId', 'updatedAt'])
        && data.classId == '6A'
        && data.day in ['Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu']
        && isValidTimeStr(data.startTime)
        && isValidTimeStr(data.endTime)
        && data.subject is string && data.subject.size() >= 2 && data.subject.size() <= 60
        && data.teacher is string && data.teacher.size() >= 2 && data.teacher.size() <= 80
        && data.room is string && data.room.size() <= 40
        && isValidId(data.authorId)
        && data.updatedAt == request.time;
    }

    function isValidAgendaItem(data) {
      return data.keys().hasAll(['classId', 'title', 'description', 'eventDate', 'eventTime', 'location', 'authorId', 'authorName', 'createdAt'])
        && data.keys().hasOnly(['classId', 'title', 'description', 'eventDate', 'eventTime', 'location', 'authorId', 'authorName', 'createdAt'])
        && data.classId == '6A'
        && data.title is string && data.title.size() >= 3 && data.title.size() <= 120
        && data.description is string && data.description.size() >= 1 && data.description.size() <= 1000
        && isValidDateStr(data.eventDate)
        && isValidTimeStr(data.eventTime)
        && data.location is string && data.location.size() >= 2 && data.location.size() <= 100
        && isValidId(data.authorId)
        && data.authorName is string && data.authorName.size() >= 2 && data.authorName.size() <= 80
        && data.createdAt is timestamp;
    }

    function isValidTreasuryEntry(data) {

export function validateProfileInput(displayName: string, absenRaw: number | string, isOwnerAccount: boolean): string | null {
  const name = displayName.trim();
  if (name.length < SCHEMA_LIMITS.displayNameMin || name.length > SCHEMA_LIMITS.displayNameMax) {
    return `Nama lengkap harus ${SCHEMA_LIMITS.displayNameMin}–${SCHEMA_LIMITS.displayNameMax} karakter.`;
  }
  const absen = typeof absenRaw === 'string' ? Number(absenRaw) : absenRaw;
  if (!Number.isInteger(absen) || absen < (isOwnerAccount ? 0 : 1) || absen > SCHEMA_LIMITS.absenMax) {
    return isOwnerAccount
      ? 'Nomor absen/kode harus bilangan bulat 0–60.'
      : 'Nomor absen siswa harus bilangan bulat 1–60.';
  }
  return null;
}

export function validateAnnouncementInput(input: {
  title: string;

  category: AnnouncementCategory;

  }
  if (!ANNOUNCEMENT_CATEGORIES.includes(input.category)) {

}

export function validateAssignmentInput(input: {

  title: string;

  if (title.length < SCHEMA_LIMITS.titleMin || title.length > SCHEMA_LIMITS.titleMax) {
    return `Judul tugas harus ${SCHEMA_LIMITS.titleMin}–${SCHEMA_LIMITS.titleMax} karakter.`;
  }
  if (description.length < SCHEMA_LIMITS.assignmentDescMin || description.length > SCHEMA_LIMITS.assignmentDescMax) {

  }
  return null;
}

export function validateScheduleInput(input: {
  day: DayOfWeek;
  startTime: string;
  endTime: string;
  subject: string;
  teacher: string;
  room: string;
}): string | null {
  if (!DAYS_ORDER.includes(input.day)) return 'Hari tidak valid.';
  if (!SCHEMA_LIMITS.timePattern.test(input.startTime) || !SCHEMA_LIMITS.timePattern.test(input.endTime)) {
    return 'Jam pelajaran harus berformat HH:MM.';
  }
  if (input.startTime >= input.endTime) {
    return 'Jam selesai harus lebih akhir dari jam mulai.';
  }
  if (input.subject.trim().length < SCHEMA_LIMITS.subjectMin || input.subject.trim().length > SCHEMA_LIMITS.subjectMax) {
    return `Mata pelajaran harus ${SCHEMA_LIMITS.subjectMin}–${SCHEMA_LIMITS.subjectMax} karakter.`;
  }

export interface TreasurySummary {
  totalIncome: number;
  totalExpense: number;
  balance: number;
}

export function calculateTreasurySummary(entries: Pick<TreasuryEntry, 'type' | 'amount'>[]): TreasurySummary {
  let totalIncome = 0;
  let totalExpense = 0;
  for (const item of entries) {
    if (item.type === 'pemasukan') {
      totalIncome += item.amount;
    } else if (item.type === 'pengeluaran') {
      totalExpense += item.amount;
    }
  }
  return {
    totalIncome,
    totalExpense,
    balance: totalIncome - totalExpense,
  };
}

export function formatRupiah(amount: number): string {
  const sign = amount < 0 ? '-' : '';
  const abs = Math.abs(Math.round(amount));
  return `${sign}Rp ${abs.toLocaleString('id-ID')}`;
}

export function toMillis(ts: unknown): number {
  if (!ts) return 0;
  if (typeof ts === 'number') return ts;
  if (ts instanceof Date) return ts.getTime();
  if (typeof ts === 'object' && ts !== null && 'toMillis' in ts && typeof (ts as { toMillis: () => number }).toMillis === 'function') {
    return (ts as { toMillis: () => number }).toMillis();
  }
  if (typeof ts === 'object' && ts !== null && 'seconds' in ts && typeof (ts as { seconds: number }).seconds === 'number') {
    return (ts as { seconds: number }).seconds * 1000;
  }
  return 0;
}

    minute: '2-digit',

}

}

function detectDeviceInfo(): string {
  const ua = typeof navigator !== 'undefined' ? navigator.userAgent : 'Unknown';
  const isAndroid = /Android/i.test(ua);
  const isIOS = /iPhone|iPad|iPod/i.test(ua);
  const isMobile = /Mobile/i.test(ua);
  const os = isAndroid ? 'Android' : isIOS ? 'iOS' : isMobile ? 'Mobile' : 'Desktop';
  const browser = /Edg/i.test(ua)
    ? 'Edge'
    : /Chrome/i.test(ua)
      ? 'Chrome'
      : /Firefox/i.test(ua)
        ? 'Firefox'
        : /Safari/i.test(ua)
          ? 'Safari'
          : 'Browser';
  return `${os} · ${browser}`.slice(0, 160);
}

export async function createOrUpdateUserProfile(
  firebaseUser: User,
  displayName: string,
  absenNumber: number,
  isFirstLoginAudit: boolean
): Promise<UserPublic> {
  const isOwnerEmail = firebaseUser.email === BOOTSTRAPPED_OWNER_EMAIL;
  const err = validateProfileInput(displayName, absenNumber, isOwnerEmail);
  if (err) throw new Error(err);

  const pubPath = `users_public/${firebaseUser.uid}`;
  try {
    const ref = doc(db, 'users_public', firebaseUser.uid);
    const existingSnap = await getDoc(ref);
    const trimmedName = displayName.trim().slice(0, 80);

    if (existingSnap.exists()) {
      const existingData = existingSnap.data() as UserPublic;
      await updateDoc(ref, {
        displayName: trimmedName,
        absenNumber,
        updatedAt: serverTimestamp(),
      });
      await syncPrivateUserOnLogin(firebaseUser);
      const updatedProfile: UserPublic = {
        ...existingData,
        displayName: trimmedName,
        absenNumber,
      };
      if (isFirstLoginAudit) {
        await recordLoginAudit(firebaseUser, updatedProfile);
      }
      return updatedProfile;
    }

    const assignedRole: UserRole = isOwnerEmail ? 'owner' : 'siswa';
    const newProfile: UserPublic = {
      uid: firebaseUser.uid,
      displayName: trimmedName,
      absenNumber,
      role: assignedRole,
      classId: CLASS_ID,
      updatedAt: serverTimestamp(),
    };
    await setDoc(ref, newProfile);
    await syncPrivateUserOnLogin(firebaseUser);
    await recordLoginAudit(firebaseUser, newProfile);
    return newProfile;
  } catch (error) {
    handleFirestoreError(error, OperationType.WRITE, pubPath);
  }
}

import React from 'react';
import { AlertTriangle, RotateCcw } from 'lucide-react';

interface State {
  hasError: boolean;
  message: string;
}

export class ErrorBoundary extends React.Component<{ children: React.ReactNode }, State> {
  constructor(props: { children: React.ReactNode }) {
    super(props);
    this.state = { hasError: false, message: '' };
  }

  static getDerivedStateFromError(error: Error): State {
    return {
      hasError: true,
      message: error.message || 'Terjadi gangguan saat memuat data kelas.',
    };
  }

  componentDidCatch(error: Error, info: React.ErrorInfo): void {
    console.error('Uncaught application error:', error, info);
  }

  render() {
    if (this.state.hasError) {
      let friendly = 'Terjadi kendala saat menghubungkan ke server kelas.';
      if (this.state.message.includes('Missing or insufficient permissions') || this.state.message.includes('permission-denied')) {
        friendly = 'Akses ditolak oleh aturan keamanan Firestore. Pastikan akun Anda telah terverifikasi dan memiliki hak akses yang sesuai.';
      }
      return (
        <div className="min-h-screen bg-[#060D1E] text-slate-100 flex items-center justify-center p-6">
          <div className="max-w-md w-full bg-[#0C1936] border border-[#1E3566] rounded-2xl p-6 space-y-4">
            <div className="flex items-center gap-3 text-amber-400">
              <AlertTriangle className="w-6 h-6 shrink-0" />
              <h1 className="text-lg font-semibold">Kendala Aplikasi</h1>
            </div>
            <p className="text-sm text-slate-300 leading-relaxed">{friendly}</p>
            <button
              type="button"
              onClick={() => window.location.reload()}
              className="w-full min-h-[44px] px-4 py-2.5 rounded-xl bg-sky-500 hover:bg-sky-400 text-[#060D1E] font-semibold text-sm flex items-center justify-center gap-2 transition-colors"

              <span>Muat Ulang Halaman</span>
            </button>
          </div>

  }
}

  profile,
  members,

  onSeedData,
  isSeeding,
}: OwnerPanelViewProps) {
  const [updatingUid, setUpdatingUid] = useState<string | null>(null);
  const [statusMessage, setStatusMessage] = useState<string | null>(null);

  if (profile.role !== 'owner') {
    return (
      <div className="bg-[#0B1733] border border-rose-500/40 rounded-2xl p-8 text-center">
        <p className="text-sm font-semibold text-rose-300">
          Halaman ini khusus untuk Owner Kelas 6A.
        </p>
      </div>
    );

    if (member.role === 'owner') return;
    const nextRole = member.role === 'admin' ? 'siswa' : 'admin';

      await updateMemberRole(member.uid, member.role, nextRole);

        `Hak akses ${member.displayName} berhasil diperbarui menjadi ${getRoleLabel(nextRole)}.`

      setUpdatingUid(null);
